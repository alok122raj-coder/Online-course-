// ── LIVE VIDEO ROOM ─────────────────────────────────────────────
function LiveRoom({ session, isTeacher, onEnd }) {
  const videoRef  = useRef(null);
  const streamRef = useRef(null);
  const [ready,   setReady]   = useState(false);
  const [camErr,  setCamErr]  = useState(null);
  const [camOn,   setCamOn]   = useState(true);
  const [micOn,   setMicOn]   = useState(true);
  const [elapsed, setElapsed] = useState(0);
  const [viewers, setViewers] = useState(Math.floor(Math.random()*12)+2);

  // Timer
  useEffect(() => {
    const t = setInterval(() => setElapsed(e=>e+1), 1000);
    return () => clearInterval(t);
  }, []);

  // Random viewer count bump (demo feel)
  useEffect(() => {
    if (!isTeacher) return;
    const t = setInterval(() => setViewers(v=>Math.max(1,v+Math.floor(Math.random()*3)-1)), 8000);
    return () => clearInterval(t);
  }, [isTeacher]);

  // Start camera (teacher only)
  useEffect(() => {
    if (!isTeacher) return;
    let mounted = true;
    (async () => {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ video:true, audio:true });
        if (!mounted) { stream.getTracks().forEach(t=>t.stop()); return; }
        streamRef.current = stream;
        if (videoRef.current) videoRef.current.srcObject = stream;
        setReady(true);
      } catch(e) {
        if (mounted) setCamErr(e.message || "Camera/mic allow nahi ki.");
      }
    })();
    return () => {
      mounted = false;
      streamRef.current?.getTracks().forEach(t=>t.stop());
    };
  }, [isTeacher]);

  const toggleCam = () => {
    const track = streamRef.current?.getVideoTracks()[0];
    if (!track) return;
    track.enabled = !track.enabled;
    setCamOn(track.enabled);
  };

  const toggleMic = () => {
    const track = streamRef.current?.getAudioTracks()[0];
    if (!track) return;
    track.enabled = !track.enabled;
    setMicOn(track.enabled);
  };

  const fmt = s => `${String(Math.floor(s/60)).padStart(2,"0")}:${String(s%60).padStart(2,"0")}`;

  return (
    <div style={{
      position:"fixed", inset:0, background:"#000", zIndex:1000,
      display:"flex", flexDirection:"column", maxWidth:480, margin:"0 auto",
    }}>
      {/* Top bar */}
      <div style={{
        padding:"14px 18px", background:"rgba(0,0,0,0.8)",
        display:"flex", alignItems:"center", justifyContent:"space-between",
        backdropFilter:"blur(10px)",
      }}>
        <div style={{display:"flex",alignItems:"center",gap:10}}>
          <div className="liveDot" style={{width:11,height:11,borderRadius:"50%",background:C.red}}/>
          <div>
            <div style={{color:"#fff",fontWeight:800,fontSize:13,lineHeight:1}}>{session?.title}</div>
            <div style={{color:C.sub,fontSize:11,marginTop:2}}>{session?.teacherName}</div>
          </div>
        </div>
        <div style={{display:"flex",alignItems:"center",gap:12}}>
          <div style={{
            display:"flex",alignItems:"center",gap:5,
            background:"rgba(255,255,255,0.08)",padding:"5px 10px",borderRadius:20,
          }}>
            <span style={{fontSize:12}}>👁</span>
            <span style={{color:"#fff",fontSize:12,fontWeight:700}}>{viewers}</span>
          </div>
          <div style={{
            background:"rgba(255,255,255,0.08)",padding:"5px 12px",borderRadius:20,
            color:"#fff",fontSize:12,fontWeight:700,fontFamily:"monospace",
          }}>
            {fmt(elapsed)}
          </div>
        </div>
      </div>

      {/* Video Area */}
      <div style={{flex:1,position:"relative",background:"#05060A",display:"flex",alignItems:"center",justifyContent:"center"}}>
        {isTeacher ? (
          <>
            {/* Teacher camera */}
            <video
              ref={videoRef}
              autoPlay muted playsInline
              style={{
                width:"100%", height:"100%", objectFit:"cover",
                display: (ready && camOn) ? "block" : "none",
              }}
            />
            {/* Camera OFF overlay */}
            {ready && !camOn && (
              <div style={{
                position:"absolute",inset:0,background:"#0a0a14",
                display:"flex",flexDirection:"column",alignItems:"center",justifyContent:"center",gap:12,
              }}>
                <span style={{fontSize:56}}>📷</span>
                <span style={{color:C.sub,fontWeight:600}}>Camera off hai</span>
              </div>
            )}
            {/* Loading */}
            {!ready && !camErr && (
              <div style={{textAlign:"center"}}>
                <div className="spin" style={{width:48,height:48,borderRadius:"50%",border:`3px solid ${C.card2}`,borderTopColor:C.orange,margin:"0 auto 16px"}}/>
                <div style={{color:C.sub,fontWeight:600}}>Camera shuru ho rahi hai…</div>
                <div style={{color:C.muted,fontSize:12,marginTop:6}}>Permission popup allow karein</div>
              </div>
            )}
            {/* Camera error */}
            {camErr && (
              <div style={{textAlign:"center",padding:28,maxWidth:300}}>
                <span style={{fontSize:52}}>🚫</span>
                <div style={{color:C.red,fontWeight:700,marginTop:14,marginBottom:8}}>Camera access nahi mila</div>
                <div style={{color:C.sub,fontSize:13,lineHeight:1.7}}>
                  Browser ke URL bar mein 🔒 icon pe click karein → Camera aur Microphone → <strong style={{color:C.text}}>Allow</strong> karein → Page refresh karein
                </div>
              </div>
            )}
            {/* Mic off indicator */}
            {ready && !micOn && (
              <div style={{
                position:"absolute",top:16,right:16,
                background:C.red,borderRadius:24,padding:"6px 14px",
                display:"flex",alignItems:"center",gap:6,
              }}>
                <span style={{fontSize:14}}>🔇</span>
                <span style={{color:"#fff",fontSize:12,fontWeight:700}}>Muted</span>
              </div>
            )}
          </>
        ) : (
          /* Student view */
          <div style={{textAlign:"center",padding:32}}>
            <div style={{
              width:100,height:100,borderRadius:"50%",
              background:`linear-gradient(135deg,${C.red}33,${C.orange}33)`,
              display:"flex",alignItems:"center",justifyContent:"center",
              margin:"0 auto 20px", border:`2px solid ${C.red}44`,
            }}>
              <span style={{fontSize:48}}>🎥</span>
            </div>
            <div style={{fontFamily:"'Playfair Display',serif",fontWeight:700,fontSize:20,marginBottom:8}}>
              {session?.title}
            </div>
            <div style={{color:C.sub,fontSize:14,marginBottom:20}}>
              {session?.teacherName} padha rahe hain
            </div>
            <div style={{
              display:"inline-flex",alignItems:"center",gap:8,
              background:`${C.red}18`,padding:"10px 22px",borderRadius:30,
              border:`1px solid ${C.red}40`,
            }}>
              <div className="blink" style={{width:9,height:9,borderRadius:"50%",background:C.red}}/>
              <span style={{color:C.red,fontWeight:800,fontSize:13}}>LIVE – {fmt(elapsed)}</span>
            </div>
            <div style={{color:C.muted,fontSize:11,marginTop:20,lineHeight:1.7,maxWidth:280,margin:"20px auto 0"}}>
              ✦ Production server ke saath teacher ki real camera stream yahan play hogi (WebRTC peer connection). Demo mode mein aap live class join kar chuke hain.
            </div>
          </div>
        )}
      </div>

      {/* Controls */}
      <div style={{
        padding:"18px 24px 32px",
        background:"rgba(0,0,0,0.9)",
        display:"flex", alignItems:"center", justifyContent:"center", gap:16,
        backdropFilter:"blur(10px)",
      }}>
        {isTeacher && (
          <>
            <CtrlBtn icon={camOn?"📷":"🚫"}  label={camOn?"Cam Off":"Cam On"}  color={camOn?C.card2:C.red}  onClick={toggleCam} />
            <CtrlBtn icon={micOn?"🎙️":"🔇"}  label={micOn?"Mute":"Unmute"}   color={micOn?C.card2:C.red}  onClick={toggleMic} />
          </>
        )}
        <CtrlBtn
          icon={isTeacher?"⏹":"👋"}
          label={isTeacher?"End Class":"Leave"}
          color={C.red}
          onClick={onEnd}
          big
        />
        {!isTeacher && (
          <div style={{color:C.sub,fontSize:12,marginLeft:4}}>Class chhod dein</div>
        )}
      </div>
    </div>
  );
}

                    <div>
                      <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:20,fontWeight:900,marginBottom:2}}>Courses</h2>
                      <p style={{color:C.sub,fontSize:13}}>{db.courses.length} courses available</p>
                    </div>
                    {isAdmin && <Btn sm onClick={()=>setShowAddCourse(true)}>+ Add Course</Btn>}
                  </div>

                  {db.courses.map(c=>(
                    <Card key={c.id} style={{marginBottom:16}}>
                      <div style={{
                        background:`linear-gradient(135deg, ${c.color}18, ${C.card2})`,
                        padding:"20px 18px", display:"flex", gap:14, alignItems:"flex-start",
                      }}>
                        <div style={{
                          width:56,height:56,borderRadius:16,
                          background:`${c.color}22`,display:"flex",alignItems:"center",
                          justifyContent:"center",fontSize:28,flexShrink:0,
                        }}>{c.emoji}</div>
                        <div style={{flex:1}}>
                          <div style={{fontWeight:800,fontSize:14,lineHeight:1.4,marginBottom:8}}>{c.title}</div>
                          <div style={{display:"flex",gap:8,flexWrap:"wrap",marginBottom:c.desc?8:0}}>
                            <Badge color={c.color}>{c.subject}</Badge>
                            <span style={{fontSize:11,color:C.sub}}>{c.teacher}</span>
                          </div>
                          {c.desc && <div style={{fontSize:12,color:C.sub,lineHeight:1.6}}>{c.desc}</div>}
                        </div>
                      </div>
                      <div style={{padding:"14px 18px"}}>
                        <Btn full color={c.color} sm onClick={()=>flash(`"${c.title}" course khul raha hai!`)}>
                          📖 Course Open Karein
                        </Btn>
                      </div>
                    </Card>
                  ))}
                </div>
              )}

              {/* ──────────── LIVE ──────────── */}
              {tab==="live" && (
                <div style={{padding:"20px 18px"}} className="fu">
                  <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:20,fontWeight:900,marginBottom:4}}>Live Classes</h2>
                  <p style={{color:C.sub,fontSize:13,marginBottom:22}}>
                    {isAdmin?"Camera se live class shuru karein":"Teacher ki live class join karein"}
                  </p>

                  {/* Live card */}
                  {db.liveSession ? (
                    <Card style={{marginBottom:20,border:`1.5px solid ${C.red}44`,background:`${C.red}0A`}}>
                      <div style={{padding:"20px 18px"}}>
                        <div style={{display:"flex",alignItems:"center",gap:8,marginBottom:12}}>
                          <div className="liveDot" style={{width:13,height:13,borderRadius:"50%",background:C.red}}/>
                          <span style={{color:C.red,fontWeight:800,fontSize:14,letterSpacing:1}}>ABHI LIVE</span>
                        </div>
                        <div style={{fontWeight:800,fontSize:17,marginBottom:4}}>{db.liveSession.title}</div>
                        <div style={{color:C.sub,fontSize:13,marginBottom:18}}>{db.liveSession.teacherName}</div>
                        {isAdmin ? (
                          <div style={{display:"flex",gap:10}}>
                            <Btn full color={C.red} onClick={()=>setInLive(true)} icon="🎥">Rejoin Live</Btn>
                            <Btn outline color={C.red} sm onClick={()=>{
                              const rec={id:`r_${Date.now()}`,title:db.liveSession.title,teacher:auth.name,date:new Date().toLocaleDateString("en-IN"),duration:"—"};
                              mutate(d=>{d.liveSession=null;d.recordings.push(rec);});
                              flash("Class khatam! Recording saved ✅");
                            }}>⏹ End</Btn>
                          </div>
                        ) : (
                          <Btn full color={C.red} onClick={joinLive} icon="🔴">Join Live Class</Btn>
                        )}
                      </div>
                    </Card>
                  ) : (
                    <Card style={{padding:24,textAlign:"center",marginBottom:20}}>
                      <span style={{fontSize:48}}>📡</span>
                      <div style={{fontWeight:800,fontSize:16,marginTop:14,marginBottom:6}}>Koi live class nahi chal rahi</div>
                      <div style={{color:C.sub,fontSize:13,marginBottom:isAdmin?20:0}}>
                        {isAdmin?"Neeche button se apni camera on karke class shuru karein":"Teacher jab live shuru kare, yahan dikhega"}
                      </div>
                      {isAdmin && <Btn onClick={()=>setShowStartLive(true)} icon="🔴">Live Class Shuru Karein</Btn>}
                    </Card>
                  )}

                  {/* Recorded classes */}
                  {db.recordings.length > 0 && (
                    <>
                      <Divider label="RECORDED CLASSES"/>
                      {db.recordings.map(r=>(
                        <Card key={r.id} style={{padding:"14px 18px",marginBottom:10}}>
                          <div style={{display:"flex",alignItems:"center",gap:12}}>
                            <div style={{width:50,height:50,borderRadius:14,background:`${C.purple}18`,display:"flex",alignItems:"center",justifyContent:"center",fontSize:24,flexShrink:0}}>🎬</div>
                            <div style={{flex:1,minWidth:0}}>
                              <div style={{fontWeight:700,fontSize:13,overflow:"hidden",textOverflow:"ellipsis",whiteSpace:"nowrap"}}>{r.title}</div>
                              <div style={{color:C.sub,fontSize:11,marginTop:2}}>{r.date} • {r.teacher}</div>
                            </div>
                            <div style={{display:"flex",gap:8}}>
                              <Btn sm onClick={()=>flash(`"${r.title}" play ho rahi hai ▶`)}>▶</Btn>
                              {isAdmin && <Btn sm outline color={C.red} onClick={()=>deleteRecording(r.id)}>🗑</Btn>}
                            </div>
                          </div>
                        </Card>
                      ))}
                    </>
                  )}
                </div>
              )}

              {/* ──────────── NOTES ──────────── */}
              {tab==="notes" && (
                <div style={{padding:"20px 18px"}} className="fu">
                  <div style={{display:"flex",justifyContent:"space-between",alignItems:"center",marginBottom:22}}>
                    <div>
                      <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:20,fontWeight:900,marginBottom:2}}>PDF Notes</h2>
                      <p style={{color:C.sub,fontSize:13}}>{db.notes.length} notes available</p>
                    </div>
                    {isAdmin && <Btn sm onClick={()=>setShowAddNote(true)}>+ Upload</Btn>}
                  </div>

                  {db.notes.map(n=>(
                    <Card key={n.id} style={{padding:"16px 18px",marginBottom:12}}>
                      <div style={{display:"flex",gap:14,alignItems:"center"}}>
                        <div style={{
                          width:52,height:64,borderRadius:12,flexShrink:0,
                          background:`${n.color}15`,
                          display:"flex",alignItems:"center",justifyContent:"center",
                          fontSize:28, border:`1px solid ${n.color}25`,
                        }}>📕</div>
                        <div style={{flex:1,minWidth:0}}>
                          <div style={{fontWeight:700,fontSize:13,lineHeight:1.4,marginBottom:6}}>{n.title}</div>
                          <div style={{display:"flex",gap:8,flexWrap:"wrap"}}>
                            <Badge color={n.color}>{n.subject}</Badge>
                            {n.pages>0 && <span style={{fontSize:11,color:C.sub}}>{n.pages} pages</span>}
                            <span style={{fontSize:11,color:C.sub}}>{n.size}</span>
                            <span style={{fontSize:11,color:C.sub}}>{n.date}</span>
                          </div>
                        </div>
                        <Btn sm outline color={C.green} onClick={()=>flash(`"${n.title}" download ho raha hai…`)} icon="⬇">
                          PDF
                        </Btn>
                      </div>
                    </Card>
                  ))}
                </div>
              )}

              {/* ──────────── PROFILE (Student) ──────────── */}
              {tab==="profile" && !isAdmin && (()=>{
                const me = db.students.find(s=>s.id===auth?.id);
                return (
                  <div style={{padding:"20px 18px"}} className="fu">
                    <h2 style={{fontFamily:"'Playfair Display',serif",fontSize:20,fontWeight:900,marginBottom:22}}>My Profile</h2>

                    {/* Profile card */}
                    <Card style={{padding:"28px 20px",textAlign:"center",marginBottom:20}}>
                      <div style={{margin:"0 auto 16px"}}>
                        <Avatar
                          photo={me?.photo}
                          name={auth?.name||"S"}
                          size={88}
                          border={me?.verified?C.green:C.yellow}
                        />
                      </div>
                      <div style={{fontFamily:"'Playfair Display',serif",fontWeight:700,fontSize:22,marginBottom:4}}>{auth?.name}</div>
                      <div style={{color:C.sub,fontSize:13,marginBottom:12}}>{me?.cls||"Class not set"}</div>
                      <Badge color={me?.verified?C.green:C.yellow}>
                        {me?.verified?"✅ Photo Verified":"⏳ Verification Pending"}
                      </Badge>
                      {!me?.verified && (
                        <div style={{color:C.muted,fontSize:11,marginTop:10,lineHeight:1.6}}>
                          Admin se request karein ki aapki photo verify karein
                        </div>
                      )}
                    </Card>

                    {/* Info rows */}
                    {[
                      ["📧 Email",     auth?.email],
                      ["📱 Phone",     me?.phone||"—"],
                      ["🎓 Class",     me?.cls||"—"],
                      ["📅 Join Date", me?.joinDate||"—"],
                      ["💳 Fee Status",me?.fee||"—"],
                    ].map(([l,v])=>(
                      <Card key={l} style={{padding:"14px 18px",marginBottom:8}}>
                        <div style={{display:"flex",justifyContent:"space-between",alignItems:"center"}}>
                          <span style={{color:C.sub,fontSize:13}}>{l}</span>
                          <span style={{fontWeight:700,fontSize:13}}>{v}</span>
                        </div>
                      </Card>
                    ))}

                    <div style={{marginTop:20}}>
                      <Btn full outline color={C.red} onClick={logout}>Logout</Btn>
                    </div>
                  </div>
                );
              })()}

            </div>{/* end page content */}

            {/* ── BOTTOM NAV ── */}
            <div style={{
              position:"fixed", bottom:0, left:"50%", transform:"translateX(-50%)",
              width:"100%", maxWidth:480,
              background:C.surface, borderTop:`1px solid ${C.border}`,
              display:"flex", padding:"8px 0 16px", zIndex:100,
              backdropFilter:"blur(16px)",
            }}>
              {tabs.map(t=>(
                <button
                  key={t.id}
                  onClick={()=>setTab(t.id)}
                  className="nav-btn"
                  style={{
                    flex:1, display:"flex", flexDirection:"column",
                    alignItems:"center", gap:4,
                    background:"none", border:"none", cursor:"pointer", padding:"6px 4px",
                    borderRadius:10, margin:"0 3px",
                  }}
                >
                  <span style={{fontSize:21}}>{t.icon}</span>
                  <span style={{
                    fontSize:9, fontWeight: tab===t.id?800:500,
                    color: tab===t.id ? C.orange : C.muted,
                    letterSpacing:0.5,
                  }}>{t.label}</span>
                  {tab===t.id && (
                    <div style={{width:18,height:3,borderRadius:2,background:C.orange}}/>
                  )}
                </button>
              ))}
            </div>
          </>
        )}

        {/* ════════ MODALS ════════ */}

        {/* Student Detail */}
        <BottomModal open={!!showStudent} onClose={()=>setShowStudent(null)} title="Student Details">
          {showStudent && (()=>{
            const s = db.students.find(x=>x.id===showStudent.id)||showStudent;
            return (
              <>
                <div style={{textAlign:"center",marginBottom:24}}>
                  <Avatar photo={s.photo} name={s.name} size={90} border={s.verified?C.green:C.yellow}/>
                  <div style={{fontFamily:"'Playfair Display',serif",fontWeight:700,fontSize:20,marginTop:14,marginBottom:4}}>{s.name}</div>
                  <Badge color={s.verified?C.green:C.yellow}>
                    {s.verified?"✅ Photo Verified":"⏳ Verify Pending"}
                  </Badge>
                </div>

                {[
                  ["Email",     s.email],
                  ["Phone",     s.phone||"—"],
                  ["Class",     s.cls||"—"],
                  ["Join Date", s.joinDate||"—"],
                  ["Fee Status",s.fee],
                ].map(([l,v])=>(
                  <div key={l} style={{display:"flex",justifyContent:"space-between",padding:"13px 0",borderBottom:`1px solid ${C.border}`}}>
                    <span style={{color:C.sub,fontSize:13}}>{l}</span>
                    <span style={{fontWeight:700,fontSize:13}}>{v}</span>
                  </div>
                ))}

                <div style={{display:"flex",flexDirection:"column",gap:10,marginTop:20}}>
                  {!s.verified && (
                    <Btn full color={C.green} onClick={()=>verifyStudent(s.id)} icon="✅">
                      Photo Verify Karein
                    </Btn>
                  )}
                  <Btn full outline color={C.yellow} onClick={()=>toggleFee(s.id)} icon="💳">
                    Fee: {s.fee==="Paid"?"Paid → Pending":"Pending → Paid"}
                  </Btn>
                </div>
              </>
            );
          })()}
        </BottomModal>

        {/* Start Live */}
        <BottomModal open={showStartLive} onClose={()=>setShowStartLive(false)} title="🔴 Live Class Shuru Karein">
          <div style={{
            background:`${C.red}10`,border:`1px solid ${C.red}25`,
            borderRadius:12,padding:"14px 16px",marginBottom:20,
          }}>
            <div style={{fontWeight:700,fontSize:13,marginBottom:4,color:C.red}}>⚠️ Important</div>
            <div style={{color:C.sub,fontSize:12,lineHeight:1.7}}>
              Browser aapki camera aur microphone access maangega. Popup mein <strong style={{color:C.text}}>Allow</strong> zaroor karein. Sab students tab live join kar sakenge.
            </div>
          </div>
          <TextInput
            label="Class Ka Title"
            placeholder="Jaise: Maths – Chapter 5, Physics Optics"
            value={liveTitle}
            onChange={e=>setLiveTitle(e.target.value)}
          />
          <Btn full color={C.red} onClick={startLive} icon="🎥">
            Camera On Karke Start Karein
          </Btn>
        </BottomModal>

        {/* Add Course */}
        <BottomModal open={showAddCourse} onClose={()=>setShowAddCourse(false)} title="Naya Course Add Karein">
          <TextInput
            label="Course Title *"
            placeholder="Jaise: Maths – Quadratic Equations"
            value={newCourse.title}
            onChange={e=>setNewCourse(x=>({...x,title:e.target.value}))}
          />
          <SelectInput
            label="Subject *"
            value={newCourse.subject}
            onChange={e=>setNewCourse(x=>({...x,subject:e.target.value}))}
            options={SUBJECTS}
            placeholder="Subject chunein"
          />
          <TextInput
            label="Description (optional)"
            placeholder="Course ke baare mein kuch likhein…"
            value={newCourse.desc}
            onChange={e=>setNewCourse(x=>({...x,desc:e.target.value}))}
          />
          <Btn full onClick={addCourse}>✓ Course Add Karein</Btn>
        </BottomModal>

        {/* Upload Note */}
        <BottomModal open={showAddNote} onClose={()=>setShowAddNote(false)} title="PDF Notes Upload Karein">
          <TextInput
            label="Notes Title *"
            placeholder="Jaise: Maths Formula Sheet Ch 1-5"
            value={newNote.title}
            onChange={e=>setNewNote(x=>({...x,title:e.target.value}))}
          />
          <SelectInput
            label="Subject *"
            value={newNote.subject}
            onChange={e=>setNewNote(x=>({...x,subject:e.target.value}))}
            options={SUBJECTS}
            placeholder="Subject chunein"
          />
          <div
            onClick={()=>noteFileRef.current.click()}
            style={{
              border:`2px dashed ${C.border}`,borderRadius:14,
              padding:"24px 20px",textAlign:"center",cursor:"pointer",
              marginBottom:18,transition:"border-color 0.2s",
            }}
            onMouseEnter={e=>e.currentTarget.style.borderColor=C.orange}
            onMouseLeave={e=>e.currentTarget.style.borderColor=C.border}
          >
            <div style={{fontSize:32,marginBottom:8}}>📁</div>
            <div style={{color:C.sub,fontSize:13}}>PDF file choose karein</div>
            <div style={{color:C.muted,fontSize:11,marginTop:4}}>(Click karein)</div>
            <input
              ref={noteFileRef} type="file" accept=".pdf"
              style={{display:"none"}}
              onChange={()=>flash("PDF select ho gaya!")}
            />
          </div>
          <Btn full onClick={addNote} icon="📄">Notes Upload Karein</Btn>
        </BottomModal>

      </div>
    </>
  );
}
